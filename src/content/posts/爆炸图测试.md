# 🧩 交互式 3D 产品爆炸图

拖动下方滑块控制爆炸程度，鼠标拖拽旋转视角，滚轮缩放。

<!-- 嵌入完整的交互式 HTML -->
<div id="explode-container" style="width:100%; height:600px; border-radius:12px; overflow:hidden; background:#111122;"></div>

<script type="importmap">
{
    "imports": {
        "three": "https://unpkg.com/three@0.160.0/build/three.module.js",
        "three/addons/": "https://unpkg.com/three@0.160.0/examples/jsm/"
    }
}
</script>

<script type="module">
import * as THREE from 'three';
import { OrbitControls } from 'three/addons/controls/OrbitControls.js';
import { CSS2DRenderer, CSS2DObject } from 'three/addons/renderers/CSS2DRenderer.js';

// --- 初始化场景、相机、渲染器 ---
const container = document.getElementById('explode-container');
const width = container.clientWidth;
const height = container.clientHeight;

const scene = new THREE.Scene();
scene.background = new THREE.Color(0x111122);

const camera = new THREE.PerspectiveCamera(45, width / height, 0.1, 1000);
camera.position.set(6, 4, 8);
camera.lookAt(0, 0, 0);

const renderer = new THREE.WebGLRenderer({ antialias: true });
renderer.setSize(width, height);
renderer.setPixelRatio(Math.min(window.devicePixelRatio, 2));
renderer.shadowMap.enabled = true;
renderer.shadowMap.type = THREE.PCFSoftShadowMap;
renderer.toneMapping = THREE.ACESFilmicToneMapping;
renderer.toneMappingExposure = 1.2;
container.appendChild(renderer.domElement);

const labelRenderer = new CSS2DRenderer();
labelRenderer.setSize(width, height);
labelRenderer.domElement.style.position = 'absolute';
labelRenderer.domElement.style.top = '0';
labelRenderer.domElement.style.left = '0';
labelRenderer.domElement.style.pointerEvents = 'none';
container.style.position = 'relative';
container.appendChild(labelRenderer.domElement);

// --- 轨道控制器 ---
const controls = new OrbitControls(camera, renderer.domElement);
controls.enableDamping = true;
controls.dampingFactor = 0.08;
controls.minDistance = 3;
controls.maxDistance = 20;
controls.target.set(0, 0, 0);
controls.update();

// --- 灯光 ---
const ambient = new THREE.AmbientLight(0x404060, 0.6);
scene.add(ambient);
const mainLight = new THREE.DirectionalLight(0xffeedd, 2.0);
mainLight.position.set(5, 8, 5);
mainLight.castShadow = true;
scene.add(mainLight);
const fillLight = new THREE.DirectionalLight(0x4488ff, 0.8);
fillLight.position.set(-3, 1, -4);
scene.add(fillLight);
const rimLight = new THREE.DirectionalLight(0xff8844, 0.5);
rimLight.position.set(-2, 3, 5);
scene.add(rimLight);

// --- 网格 ---
const grid = new THREE.GridHelper(10, 20, 0x446688, 0x223355);
grid.position.y = -1.2;
scene.add(grid);

// --- 部件数据 ---
const parts = [];
const originalPositions = [];
const explodeDirections = [];

function createPart(geometry, color, pos, name, emissive = 0x000000, emissiveIntensity = 0) {
    const mat = new THREE.MeshPhysicalMaterial({
        color, metalness: 0.3, roughness: 0.4, clearcoat: 0.1,
        emissive, emissiveIntensity, envMapIntensity: 0.6
    });
    const mesh = new THREE.Mesh(geometry, mat);
    mesh.position.copy(pos);
    mesh.castShadow = true;
    mesh.receiveShadow = true;
    // 边框
    const edges = new THREE.EdgesGeometry(geometry);
    const wire = new THREE.LineSegments(edges, new THREE.LineBasicMaterial({ color: 0x88ccff, transparent: true, opacity: 0.25 }));
    mesh.add(wire);
    // 标签
    const div = document.createElement('div');
    div.textContent = name;
    div.style.color = '#aaccff';
    div.style.fontSize = '14px';
    div.style.fontWeight = 'bold';
    div.style.textShadow = '0 0 15px rgba(0,0,0,0.9)';
    div.style.background = 'rgba(10,10,30,0.6)';
    div.style.padding = '4px 12px';
    div.style.borderRadius = '20px';
    div.style.border = '1px solid rgba(100,180,255,0.2)';
    div.style.backdropFilter = 'blur(4px)';
    div.style.pointerEvents = 'none';
    div.style.userSelect = 'none';
    const label = new CSS2DObject(div);
    label.position.set(0, 0.8, 0);
    mesh.add(label);
    scene.add(mesh);
    parts.push(mesh);
    originalPositions.push(pos.clone());
    return mesh;
}

// 创建各个部件
createPart(new THREE.BoxGeometry(1.2, 1.2, 1.2), 0x4facfe, new THREE.Vector3(0,0,0), '⚙️ 核心', 0x4facfe, 0.15);
createPart(new THREE.CylinderGeometry(1.0, 1.0, 0.4, 32), 0xff6b6b, new THREE.Vector3(0,1.5,0), '🔴 顶盖');
createPart(new THREE.CylinderGeometry(1.4, 1.6, 0.5, 32), 0x51cf66, new THREE.Vector3(0,-1.5,0), '🟢 底座');
createPart(new THREE.BoxGeometry(0.8, 0.6, 1.8), 0xfcc419, new THREE.Vector3(-2.0,0.2,0), '🟡 左翼');
createPart(new THREE.BoxGeometry(0.8, 0.6, 1.8), 0xfcc419, new THREE.Vector3(2.0,0.2,0), '🟡 右翼');
createPart(new THREE.SphereGeometry(0.6, 24, 24), 0xcc5de8, new THREE.Vector3(0,0.1,1.8), '🟣 前盖');
createPart(new THREE.SphereGeometry(0.6, 24, 24), 0xcc5de8, new THREE.Vector3(0,0.1,-1.8), '🟣 后盖');

// 装饰环（特殊处理）
const ringGeo = new THREE.TorusGeometry(0.9, 0.12, 16, 48);
const ringMat = new THREE.MeshPhysicalMaterial({ color: 0xffd43b, metalness: 0.7, roughness: 0.2, emissive: 0xffd43b, emissiveIntensity: 0.05 });
const ring = new THREE.Mesh(ringGeo, ringMat);
ring.position.set(0, 0.1, 0);
ring.rotation.x = Math.PI / 2;
ring.castShadow = true;
ring.receiveShadow = true;
const ringDiv = document.createElement('div');
ringDiv.textContent = '💫 装饰环';
ringDiv.style.cssText = 'color:#ffd43b;font-size:14px;font-weight:bold;text-shadow:0 0 15px rgba(0,0,0,0.9);background:rgba(10,10,30,0.5);padding:4px 12px;border-radius:20px;border:1px solid rgba(255,212,59,0.2);backdrop-filter:blur(4px);pointer-events:none;user-select:none;';
const ringLabel = new CSS2DObject(ringDiv);
ringLabel.position.set(0, 1.0, 0);
ring.add(ringLabel);
scene.add(ring);
parts.push(ring);
originalPositions.push(new THREE.Vector3(0, 0.1, 0));

// 计算爆炸方向
parts.forEach((p, i) => {
    const dir = originalPositions[i].clone();
    if (dir.length() < 0.01) dir.set(0, 1, 0);
    else dir.normalize();
    explodeDirections.push(dir);
});

// --- 爆炸控制 UI ---
const controlsDiv = document.createElement('div');
controlsDiv.style.cssText = 'position:absolute;bottom:30px;left:50%;transform:translateX(-50%);background:rgba(30,30,40,0.8);backdrop-filter:blur(8px);padding:14px 30px;border-radius:50px;display:flex;gap:25px;align-items:center;color:#eee;border:1px solid rgba(255,255,255,0.15);box-shadow:0 8px 30px rgba(0,0,0,0.5);z-index:20;pointer-events:auto;';
container.appendChild(controlsDiv);

const labelLeft = document.createElement('span');
labelLeft.textContent = '⬇️ 组合';
labelLeft.style.color = '#aaa';
controlsDiv.appendChild(labelLeft);

const slider = document.createElement('input');
slider.type = 'range';
slider.min = 0;
slider.max = 1;
slider.step = 0.001;
slider.value = 0;
slider.style.cssText = 'width:200px;height:4px;-webkit-appearance:none;appearance:none;background:linear-gradient(90deg,#4facfe,#00f2fe);border-radius:2px;outline:none;cursor:pointer;';
slider.addEventListener('input', (e) => {
    const t = parseFloat(e.target.value);
    parts.forEach((part, idx) => {
        const orig = originalPositions[idx];
        const dir = explodeDirections[idx];
        const target = orig.clone().add(dir.clone().multiplyScalar(2.8 * t));
        part.position.copy(target);
    });
});
controlsDiv.appendChild(slider);

const labelRight = document.createElement('span');
labelRight.textContent = '爆炸 ⬆️';
labelRight.style.color = '#aaa';
controlsDiv.appendChild(labelRight);

const resetBtn = document.createElement('button');
resetBtn.textContent = '⟲ 重置视角';
resetBtn.style.cssText = 'background:rgba(255,255,255,0.12);border:1px solid rgba(255,255,255,0.25);color:#fff;padding:6px 18px;border-radius:30px;cursor:pointer;font-size:14px;transition:all 0.2s;';
resetBtn.addEventListener('click', () => {
    camera.position.set(6, 4, 8);
    controls.target.set(0, 0, 0);
    controls.update();
});
controlsDiv.appendChild(resetBtn);

// 自适应
function resizeRenderer() {
    const w = container.clientWidth;
    const h = container.clientHeight;
    camera.aspect = w / h;
    camera.updateProjectionMatrix();
    renderer.setSize(w, h);
    labelRenderer.setSize(w, h);
}
window.addEventListener('resize', resizeRenderer);

// 动画循环
function animate() {
    requestAnimationFrame(animate);
    controls.update();
    renderer.render(scene, camera);
    labelRenderer.render(scene, camera);
}
animate();

// 初始组合状态
parts.forEach((p, i) => {
    p.position.copy(originalPositions[i]);
});
slider.value = 0;

console.log('✅ 交互式爆炸图已加载！');
</script>

## 使用说明

- **拖动滑块**：在底部滑块上左右拖动，即可控制部件从组合到爆炸的平滑过渡。
- **旋转视角**：按住鼠标左键拖拽，可以任意角度观察产品。
- **缩放**：滚动滚轮拉近拉远。
- **重置视角**：点击“重置视角”按钮回到默认角度。

## 自定义模型

如需替换为自己的产品模型，可以修改 `createPart` 调用的几何体、颜色和位置，或者直接加载外部 `.glb`/`.gltf` 文件（需额外代码）。

## 运行环境要求

- 支持内联 HTML 和 ES Module 的 Markdown 编辑器（如 **Typora**、**Obsidian** 开启“允许脚本”）。
- 或者将本文件保存为 `.html`，直接用浏览器打开（此时 Markdown 标题会被忽略，但交互内容正常显示）。
- 需要联网加载 Three.js 库。
